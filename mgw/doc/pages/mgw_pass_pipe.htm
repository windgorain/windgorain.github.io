pass_pipe

作用

忽略下游实例返回值, 并返回PASS给上游实例

MGW文件

pipe/xdp_pass_pipe_kern.o

pipe/tc_pass_pipe_kern.o

控制程序

无

示例脚本


#/bin/bash
#文件名称: xdp_pass_pipe.sh
#使用方法: ./xdp_pass_pipe.sh 接口
#使用示例: ./xdp_pass_pipe.sh eth0
#终止脚本: Ctrl+C

IF_NAME=eth0
if [ "$1" ];then
    IF_NAME=$1
fi

#加载mgw
./mgwtool load instance -n pass -f pipe/xdp_pass_pipe_kern.o
./mgwtool load instance -n drop -f leaf/xdp_drop_leaf_kern.o

#关联到接口
./mgwtool attach xdp -n pass -t ${IF_NAME} -p mgw_input

#关联到slot
./mgwtool attach instance -n drop -t pass -p mgw_input

./mgwtool enable all

trap finish INT
finish () {
    echo ""
    echo "quit..."
    #从接口解除关联
    ./mgwtool deta xdp -t ${IF_NAME} 
    #卸载所有instance
    ./mgwtool unload all-instance
    exit
}

sleep infinity
