cpu_traffic

中文名

CPU流量控制器

作用

根据cpu id调度流量, 将流量调度到和cpu id数字相同的slot上

MGW文件

traffic/xdp_cpu_traffic_kern.o

traffic/tc_cpu_traffic_kern.o

控制程序

无

示例脚本


#/bin/bash
#文件名称: xdp_cpu_traffic.sh
#使用方法: ./xdp_cpu_traffic.sh 接口
#使用示例: ./xdp_cpu_traffic.sh eth0
#终止脚本: Ctrl+C

IF_NAME=eth0
if [ "$1" ];then
    IF_NAME=$1
fi

#加载 traffic
./mgwtool load instance -n t1 -f traffic/xdp_cpu_traffic_kern.o

CPU_NUM=`grep -c "model name" /proc/cpuinfo`
CPU_NUM=$(($CPU_NUM - 1))

for i in $(seq 0 ${CPU_NUM})
  do
    echo "load p${i}"
    ./mgwtool load instance -n p${i} -f leaf/xdp_ip_printer_leaf_kern.o
    ./mgwtool attach instance -n p${i} -t t1 -p mgw_input -s ${i}
  done


#将t1关联到接口
./mgwtool attach xdp -n t1 -t ${IF_NAME} -p mgw_input

./mgwtool enable all

trap finish INT
finish () {
    echo ""
    echo "quit..."
    #从接口解除关联
    ./mgwtool deta xdp -t ${IF_NAME} 
    #卸载所有instance
    ./mgwtool unload all-instance
    exit
}
