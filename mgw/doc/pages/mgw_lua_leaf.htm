lua_leaf

作用

执行Lua脚本

需求

需要KLC环境支持

MGW文件

leaf/xdp_lua_leaf_klc.o

leaf/tc_lua_leaf_klc.o

控制程序

leaf/mgw-lua

示例脚本


#/bin/bash
#文件名称: xdp_lua.sh
#使用方法: ./xdp_lua.sh 接口
#使用示例: ./xdp_lua.sh eth0
#加载lua: ./klctool load spf lua5.3.spf
#终止脚本: Ctrl+C

IF_NAME=eth0
if [ "$1" ];then
    IF_NAME=$1
fi

#加载mgw 
./mgwtool load instance -n lua1 -f leaf/xdp_lua_leaf_kern.o

#将n1关联到接口
./mgwtool attach xdp -n lua1 -t ${IF_NAME} -p mgw_input

./mgwtool enable all

trap finish INT
finish () {
    echo ""
    echo "quit..."
    #从接口解除关联
    ./mgwtool deta xdp -t ${IF_NAME} 
    #卸载所有instance
    ./mgwtool unload all-instance
    exit
}
