dport_traffic

作用

根据 目的端口/内存报文目的端口 进行流量导流

MGW文件

traffic/xdp_dport_traffic_kern.o

traffic/xdp_vxlan_inner_dport_traffic_kern.o

traffic/xdp_geneve_inner_dport_traffic_kern.o

traffic/tc_dport_traffic_kern.o

traffic/tc_vxlan_inner_dport_traffic_kern.o

traffic/tc_geneve_inner_dport_traffic_kern.o

控制程序

traffic/mgw-port-traffic

示例脚本


#/bin/bash
#文件名称: xdp_dport_traffic.sh
#使用方法: ./xdp_dport_traffic.sh 接口
#使用示例: ./xdp_dport_traffic.sh eth0
#终止脚本: Ctrl+C

IF_NAME=eth0
if [ "$1" ];then
    IF_NAME=$1
fi

#加载 xdp ip traffic
./mgwtool load instance -n t1 -f traffic/xdp_dport_traffic_kern.o
#加载 xdp ip printer
./mgwtool load instance -n p1 -f leaf/xdp_ip_printer_leaf_kern.o
./mgwtool load instance -n p2 -f leaf/xdp_ip_printer_leaf_kern.o
#将t1关联到接口
./mgwtool attach xdp -n t1 -t ${IF_NAME} -p mgw_input
#将p1/p2关联到t1的slot 0/1
./mgwtool attach instance -n p1 -t t1 -p mgw_input -s 0
./mgwtool attach instance -n p2 -t t1 -p mgw_input -s 1
#为t1添加导流规则: 将流量分别导流向slot 0/1
./traffic/mgw-port-traffic add rule -n t1 -p 80 -s 0
./traffic/mgw-port-traffic add rule -n t1 -p 443 -s 1

./mgwtool enable all

trap finish INT
finish () {
    echo ""
    echo "quit..."
    #从接口解除关联
    ./mgwtool detach xdp -t ${IF_NAME} 
    #卸载所有instance
    ./mgwtool unload all-instance
    exit
}
