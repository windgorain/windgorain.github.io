ip_cluster

作用

根据 IP和端口/内层报文的IP和端口 进行一致性Hash并分发到下游实例

MGW文件

cluster/xdp_ip_cluster_kern.o

cluster/xdp_vxlan_inner_ip_cluster_kern.o

cluster/xdp_geneve_inner_ip_cluster_kern.o

cluster/tc_ip_cluster_kern.o

cluster/tc_vxlan_inner_ip_cluster_kern.o

cluster/tc_geneve_inner_ip_cluster_kern.o

控制程序

无

示例脚本: 


#/bin/bash
#文件名称: xdp_ip_cluster.sh
#使用方法: ./xdp_ip_cluster.sh 接口
#使用示例: ./xdp_ip_cluster.sh eth0
#终止脚本: Ctrl+C

IF_NAME=eth0
if [ "$1" ];then
    IF_NAME=$1
fi

#加载 cluster
./mgwtool load instance -n c1 -f cluster/xdp_ip_cluster_kern.o

#加载 xdp ip printers, 并关联到c1
for i in $(seq 1 4)
  do
    ./mgwtool load instance -n p${i} -f leaf/xdp_ip_printer_leaf_kern.o
    ./mgwtool attach instance -n p${i} -t c1 -p mgw_input
  done

#将c1关联到接口
./mgwtool attach xdp -n c1 -t ${IF_NAME} -p mgw_input

./mgwtool enable all

trap finish INT
finish () {
    echo ""
    echo "quit..."
    #从接口解除关联
    ./mgwtool deta xdp -t ${IF_NAME}
    #卸载所有instance
    ./mgwtool unload all-instance
    exit
}
